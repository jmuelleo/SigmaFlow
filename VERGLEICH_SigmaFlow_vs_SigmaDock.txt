python
import os, json
from pymol import cmd, util

BASE = r"C:\Users\julia\Documents\SigmaFlow\SigmaFlow_Variants\posebusters_full_comparison\pymol_cases"

FAELLE = json.load(open(os.path.join(BASE, 'cases.json'), encoding='utf-8'))

def zeige(cid, protein=True, oberflaeche=True):
    """Zeigt Protein + wahren Liganden + SigmaFlow + SigmaDock fuer einen Komplex."""
    cmd.delete('all')
    cmd.load(os.path.join(BASE, cid + '_TRUE.sdf'),      'LIG_WAHR')
    cmd.load(os.path.join(BASE, cid + '_SIGMAFLOW.sdf'), 'LIG_SIGMAFLOW')
    cmd.load(os.path.join(BASE, cid + '_SIGMADOCK.sdf'), 'LIG_SIGMADOCK')
    if protein:
        try:
            cmd.fetch(cid.split('_')[0], 'PROT', type='pdb', async_=0)
            cmd.remove('PROT and not polymer')      # Wasser, Ionen, Kristallliganden weg
            cmd.remove('PROT and hydro')
        except Exception as e:
            print('[WARN] Protein konnte nicht geladen werden:', e)
    cmd.hide('everything')
    cmd.show('sticks', 'LIG_*')
    util.cbag('LIG_WAHR')          # Kohlenstoff gruen   = Kristallpose
    util.cbac('LIG_SIGMAFLOW')     # Kohlenstoff cyan    = SigmaFlow
    util.cbam('LIG_SIGMADOCK')     # Kohlenstoff magenta = SigmaDock
    if protein and cmd.count_atoms('PROT'):
        cmd.show('cartoon', 'PROT')
        cmd.color('grey70', 'PROT')
        cmd.set('cartoon_transparency', 0.5)
        if oberflaeche:
            cmd.select('TASCHE', 'byres (PROT within 6 of LIG_WAHR)')
            cmd.show('surface', 'TASCHE')
            cmd.set('transparency', 0.55)
            cmd.color('wheat', 'TASCHE')
            cmd.deselect()
    cmd.bg_color('white')
    cmd.set('ray_opaque_background', 0)
    cmd.set('stick_radius', 0.14)
    cmd.set('valence', 1)
    cmd.orient('LIG_WAHR')
    cmd.zoom('LIG_WAHR', 6)
    m = FAELLE.get(cid, {})
    print('=' * 78)
    print(f"{cid}  [{m.get('set','?')}]  {m.get('rolle','')}")
    print(f"  RMSD SigmaFlow {m.get('rmsd_sf','?')} A | SigmaDock {m.get('rmsd_sd','?')} A"
          f"  | Orientierungsfehler SigmaFlow {m.get('orient_sf','?')} Grad")
    print('  gruen = Kristallpose | cyan = SigmaFlow | magenta = SigmaDock')
    if m.get('warnung'):
        print('  [!] ' + m['warnung'])
    print('=' * 78)

def liste():
    for s in sorted({v.get('set','?') for v in FAELLE.values()}):
        print(f'--- {s} ---')
        for k, v in FAELLE.items():
            if v.get('set') != s:
                continue
            print(f"  zeige(\"{k}\")   {v.get('rolle','')}: "
                  f"SF {v.get('rmsd_sf','?')} A / SD {v.get('rmsd_sd','?')} A, "
                  f"Orient. {v.get('orient_sf','?')} Grad")

cmd.extend('zeige', zeige)
cmd.extend('liste', liste)
python end
